# R35S — símbolos que o kernel PRECISA ter, venha de onde vier
#
# O `linux.fragment` é um DELTA: ele pede o que o defconfig da arquitetura não
# liga. Símbolo que o defconfig arm64 já traz (`CONFIG_MMC_DW_ROCKCHIP`,
# `CONFIG_EXT4_FS`, `CONFIG_SERIAL_8250_DW`…) NÃO está lá, e se sumir ninguém
# percebe até o aparelho não bootar.
#
# Este arquivo é a outra metade do gate: `assert-kernel-config --required` falha
# o build se algum destes desaparecer ou cair para `=m` sem justificativa.
#
# Formato: um `CONFIG_*` por linha. `#` comenta.

# Sem isto o cartão não é lido e não há rootfs.
CONFIG_MMC_DW_ROCKCHIP=y

# Sem isto não há sistema de arquivos no cartão.
CONFIG_EXT4_FS=y
CONFIG_VFAT_FS=y
CONFIG_TMPFS=y

# Console serial: é o ÚNICO canal de diagnóstico desta placa (sem rádio onboard,
# e o `dwc2` nasce como host). Medido: ttyS2 a 1500000 em 0xff160000.
CONFIG_SERIAL_8250=y
CONFIG_SERIAL_8250_DW=y
CONFIG_SERIAL_8250_CONSOLE=y

# DRM/DSI: o painel é DSI e o PHY tem de ser `=y` (como módulo dá EPROBE_DEFER e
# tela preta, sem mensagem ligando a causa).
CONFIG_DRM=y
CONFIG_DRM_ROCKCHIP=y
CONFIG_DRM_DW_MIPI_DSI=y
CONFIG_PHY_ROCKCHIP_INNO_DSIDPHY=y
# GPU: pedido no fragmento, exigido aqui. Sem este símbolo o device tree declara
# `gpu@ff400000` e o kernel não tem quem atenda — o sintoma é `softpipe` em
# `GL_RENDERER` com `GL_VERSION` dizendo 3.1, ou seja, 3D na CPU e todo teste
# superficial aprovando.
CONFIG_DRM_PANFROST=y

# Entrada: 16 botões via multiplexador de ADC + 2 eixos do stick esquerdo.
CONFIG_INPUT_EVDEV=y
CONFIG_JOYSTICK_ADC=y
CONFIG_ROCKCHIP_SARADC=y

# Energia e térmica. O DT declara o `charger` do rk817 (com `monitored-battery`)
# e as `thermal-zones`; sem estes símbolos o kernel ignora os dois em silêncio e
# o aparelho fica sem nível de bateria e sem proteção térmica. Medido no R35S em
# 06/11/2026: `modprobe` de cada um criou `rk817-battery` e `thermal_zone0/1`,
# provando que o hardware responde — faltava o driver no boot.
CONFIG_CHARGER_RK817=y
CONFIG_THERMAL=y
CONFIG_ROCKCHIP_THERMAL=y

# O driver que de fato atende o nó `rocknix-singleadc-joypad` do DTS. Não é do
# mainline: vem do patch `0002-giochi-joypad-singleadc.patch`. Se o patch parar
# de aplicar, este símbolo some — e o `assert-kernel-config` falha o build em vez
# de entregar uma imagem sem botão nenhum.
CONFIG_JOYSTICK_ODROIDGO3=y

# USB: canal de shell (RNDIS) e a única porta para dongle de rádio.
CONFIG_USB_GADGET=y
CONFIG_USB_CONFIGFS=y
CONFIG_USB_DWC2=y

# Áudio: card `rk817_int`, codec no PMIC. Medido no aparelho.
CONFIG_SND_SOC_RK817=y
CONFIG_MFD_RK8XX=y

# O driver que atende o nó `panel@0` do DTS. Não é do mainline: vem do patch
# `0003-giochi-painel-generic-dsi.patch`. Se o patch parar de aplicar, este
# símbolo some e o build FALHA — em vez de entregar imagem com tela preta e
# sistema bootando, que é indistinguível de "não ligou".
CONFIG_DRM_PANEL_GENERIC_DSI=y

# O card que lê o nó `rk817-sound` do DTS. Medido em 06/11/2026: sem ele, codec
# e I2S `=y` e mesmo assim `--- no soundcards ---`, porque o módulo existia na
# imagem e nada o carregava. O boot é a única prova de que ele veio.
CONFIG_SND_SIMPLE_CARD=y
